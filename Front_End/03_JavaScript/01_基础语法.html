JavaScript 基础语法

JavaScript（JS）是网页的编程语言，用于实现网页的交互功能。 前面的 HTML 是"静态的描述"，只能写死内容；JS 让网页可以响应用户操作、动态改变自己。

技术负责典型动作
HTML网页结构这里有个按钮
CSS网页样式按钮是蓝色的圆角
JavaScript网页行为点一下按钮，数字加一
本文件的输出方式：课案里所有示例都用 alert("...") 弹窗输出。 弹窗会打断浏览（而且弹太多次很烦），所以本文件改成把结果输出到页面上的 绿色「输出面板」。注释里会同时给出 alert 的原始写法。 你自己练习时两种都可以用，调试时更推荐 console.log() + F12 的 Console 面板。

一、变量

变量就是「给一个值起个名字」，方便后面反复使用。 现代 JS 用 let（可重新赋值）和 const（不可重新赋值）。 

let name = "张三";        // 字符串
let age = 25;             // 数字
let isStudent = true;     // 布尔值

// 课案原文（alert 版）：
// alert("姓名: " + name);
// alert("年龄: " + age);
// alert("是学生吗: " + isStudent);

console.log(name, age, isStudent);   // 更推荐：打到控制台，不打断浏览
↓ 输出
关键字能否重新赋值作用域建议
const❌ 不能块级 {}默认用它，能不变就不变
let✅ 能块级 {}需要改值时用它
var✅ 能函数级（会"变量提升"）不要用，是历史遗留写法
const PI = 3.14159;
// PI = 3;              // ❌ 报错：Assignment to constant variable.

let count = 0;
count = count + 1;      // ✅ 可以

// 关于 const 的一个坑：const 锁的是"变量指向"，不是"内容"
const arr = [1, 2, 3];
arr.push(4);            // ✅ 可以！数组内容变了，但 arr 还是指向同一个数组
// arr = [5, 6];        // ❌ 报错：不能让它指向别的数组
未声明的变量会给全局对象加属性，是 bug 的温床。永远写 let / const。 （本文件用 "use strict" 之外的普通模式演示，你不小心写漏关键字时浏览器不报错， 但用 "use strict"; 开头就会立即报错——这是推荐做法。）

二、数据类型

JS 是弱类型 / 动态类型语言：变量的类型由它当前装的值决定，不需要提前声明类型。

let str   = "Hello";                 // 1. 字符串 string（单引号/双引号/反引号都行）
let num   = 25;                      // 2. 数字 number（整数和小数都是 number）
let bool  = true;                    // 3. 布尔 boolean（只有 true / false）
let arr   = [1, 2, 3];               // 4. 数组 array
let obj   = { name: "张三" };         // 5. 对象 object（键值对）
let nothing = null;                  // 6. null：空值（"我故意让它为空"）
let notDefined;                      // 7. undefined：未定义（声明了但没赋值）

// 课案原文（alert 版）：

// alert("数字: " + num);
// alert("布尔值: " + bool);
// alert("数组: " + arr);
// alert("对象: " + obj.name);   // 注意：obj.name 取的是对象里的 name 属性

字符串模板（反引号）—— 最实用的语法糖

const name = "张三", age = 25;
// 老办法：用 + 拼字符串，容易漏空格、引号难嵌套
console.log("姓名: " + name + ", 年龄: " + age);
// 新办法：反引号 + ${} 插值
console.log(`姓名: ${name}, 年龄: ${age}`);
console.log(`明年他 ${age + 1} 岁`);     // ${} 里可以写任意表达式

三、运算符

算术运算符

+ - * / %（取余） **（幂）；++ / -- 自增自减。 

console.log(7 + 3);      // 10
console.log(7 - 3);      // 4
console.log(7 * 3);      // 21
console.log(7 / 3);      // 2.3333333333333335（JS 的除法永远是小数，不会整除）
console.log(7 % 3);      // 1  （取余数，判断奇偶常用）
console.log(2 ** 10);    // 1024
console.log("5" + 3);    // "53"  ⚠️ + 遇到字符串会变成"拼接"
console.log("5" - 3);    // 2     ⚠️ - 会把字符串强转成数字

比较运算符（重点：== 与 ===）

console.log(5 ==  "5");   // true   ← 只比"值"，会自动做类型转换（不推荐）
console.log(5 === "5");   // false  ← 比"值和类型"，都相同才 true（推荐）
console.log(1 ==  true);  // true   ← 因为 true 被转成了 1
console.log(0 ==  false); // true
console.log(null == undefined);   // true
console.log(NaN == NaN);          // false  ← NaN 不等于自己！
console.log(Object.is(NaN, NaN)); // true
规矩：一律用 === 和 !==，永远不要用 == 和 !=。 == 的类型转换规则极其反直觉（比如 [] == false 是 true）， 是 JS 最经典的 bug 来源之一。

逻辑运算符

console.log(true && false);   // false  —— 与：两个都真才真
console.log(true || false);   // true   —— 或：有一个真就真
console.log(!true);           // false  —— 非：取反

// 短路求值（很常用）：
const user = null;
const nickname = user && user.name;        // user 是 null，直接得到 null，不会报错
const display  = nickname || "匿名用户";    // nickname 为假，取默认值

// ?? 空值合并：只在 null / undefined 时取默认值（比 || 更精确）
const n = 0;
console.log(n || 100);    // 100  ⚠️ 0 被视为假，不符合预期
console.log(n ?? 100);    // 0    ✅ 只有 null/undefined 才替换
// ?. 可选链：如果左边是 null/undefined，直接返回 undefined，不报错
console.log(user?.name);  // undefined（不会抛异常）
↓ 输出

四种循环对比

写法拿到什么适合
for (let i=0; i<n; i++)下标 i需要知道序号、需要中断
for (const item of arr)值遍历数组的首选
for (const key in obj)键名（字符串）遍历对象的属性
while (条件) {}—循环次数不确定
do { } while (条件)—至少执行一次
// 数组的高阶方法（比 for 循环更"声明式"，实际开发中用得更多）
const nums = [1, 2, 3, 4, 5];

// forEach：逐个处理，没有返回值
nums.forEach(function (n) { console.log(n); });

// map：把每个元素"映射"成新值，返回新数组（长度不变）
const doubled = nums.map(n => n * 2);           // [2, 4, 6, 8, 10]

// filter：筛选，返回长度可能变小的新数组
const evens = nums.filter(n => n % 2 === 0);    // [2, 4]

// reduce：把数组"归约"成一个值（求和、求最大值等）
const sum = nums.reduce((acc, n) => acc + n, 0);  // 15

// find：找第一个满足条件的元素，找不到返回 undefined
const found = nums.find(n => n > 3);            // 4

// some / every：是否存在 / 是否全部满足
console.log(nums.some(n => n > 4));    // true
console.log(nums.every(n => n > 0));   // true

// 注意：map/filter/reduce 都返回【新数组】，不会修改原数组（不改变原数据 = 更安全）
↓ 输出
// while 循环：数到 5
let i = 0;
while (i < 5) {
    console.log(i);
    i++;            // ⚠️ 忘了这行就是"死循环"，浏览器会卡死
}

// break / continue
for (let j = 0; j < 10; j++) {
    if (j === 3) continue;   // 跳过本轮剩下的代码，直接进入下一轮
    if (j === 6) break;      // 直接跳出整个循环
    console.log(j);          // 输出 0 1 2 4 5
}

四种定义函数的方式

// ① 函数声明（会"提升"，可以先调用后定义）
function add(a, b) { return a + b; }

// ② 函数表达式（不会提升，必须先定义后调用）
const add2 = function (a, b) { return a + b; };

// ③ 箭头函数（ES6 新增，最简洁，没有自己的 this）
const add3 = (a, b) => a + b;                 // 只有一行 return 时可以省略 {} 和 return
const add4 = (a, b) => { return a + b; };     // 写了 {} 就必须显式 return
const square = x => x * x;                    // 只有一个参数可省略括号
const sayHi = () => console.log("Hi");        // 没有参数必须写 ()

// ④ 立即执行函数（旧写法，现代基本不用）
(function () { console.log("马上执行"); })();

参数与返回值

// 默认参数
function greet(name = "陌生人", greeting = "你好") {
    return `${greeting}，${name}！`;
}
greet();                    // "你好，陌生人！"
greet("张三");               // "你好，张三！"
greet("张三", "早上好");      // "早上好，张三！"

// 剩余参数（把多余的实参收进一个真数组）
function sum(...nums) {
    return nums.reduce((a, b) => a + b, 0);
}
sum(1, 2, 3, 4);            // 10

// 没有 return 的函数，返回值是 undefined
function noReturn() { const x = 1; }
console.log(noReturn());    // undefined

作用域与闭包（进阶，知道概念即可）

let globalVar = "我是全局的";       // 函数外定义 → 到处都能访问

function outer() {
    let outerVar = "我是 outer 的局部变量";

    function inner() {
        // 内层函数可以访问外层的变量，这叫「作用域链」
        console.log(outerVar);
        console.log(globalVar);
    }
    return inner;
}
const fn = outer();     // outer 已经执行完了
fn();                   // 但依然能打印 outerVar —— 这就是「闭包」
闭包的实际用途：「记住」状态。例如做一个计数器， 每次调用都加一，而不需要全局变量 —— 见本页最后一个演示。

<script> 放哪里？

位置问题 / 好处
<head> 里 ❌ 此时 <body> 还没解析，getElementById 会拿到 null 而报错
</body> 前（本文件的做法） ✅ DOM 已经准备好了，最稳妥的传统做法
<head> + defer 属性 ✅ 现代推荐：<script src="app.js" defer></script>， 下载不阻塞渲染，且保证在 DOM 解析完成后按顺序执行
最常见的 JS 报错：Uncaught TypeError: Cannot read properties of null (reading 'onclick') —— 99% 是因为 <script> 写在 <head> 里、执行时元素还不存在。 解决办法：把 script 移到 </body> 前，或者加 defer。

八、动手练习

把下面这段代码抄进一个新的 .html 文件里跑一跑（或者直接改本文件）：

// 用闭包做一个计数器工厂
function createCounter() {
    let count = 0;                  // 这个变量被下面的函数"记住"了
    return function () {
        count += 1;
        return count;
    };
}

const c1 = createCounter();
const c2 = createCounter();         // 每个计数器有自己独立的 count
console.log(c1());   // 1
console.log(c1());   // 2
console.log(c2());   // 1  ← 互不影响
