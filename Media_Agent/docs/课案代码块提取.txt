"""工作流基类 — 提取公共逻辑（与 workflows/__init__.py 功能一致，别名兼容）"""
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage
from config import settings

def get_model(temperature: float = 0.5) -> ChatOpenAI:
    """获取模型实例（统一配置，含超时避免挂死）"""
    return ChatOpenAI(
        api_key=settings.api_key,
        base_url=settings.base_url,
        model=settings.model_name,
        temperature=temperature,
        request_timeout=90,    # 90秒超时（长prompt+复杂推理需要更久）
        max_retries=1,         # 失败重试1次
    )

def safe_llm_call(model: ChatOpenAI, prompt: str, fallback: str = "") -> str:
    """安全调用LLM，带错误处理。失败时返回错误信息，绝不抛异常。"""
    try:
        response = model.invoke([HumanMessage(content=prompt)])
        return response.content
    except Exception as e:
        return f"[LLM调用失败: {e}]" if not fallback else fallback

# 别名兼容（供 workflows/__init__.py 的使用者）
llm_call = safe_llm_call

# 文件: workflows/positioning.py
"""账号定位工作流 — 3步：画像分析 → 对标搜索 → 方案生成"""
from typing import TypedDict
from langgraph.graph import StateGraph, START, END
from workflows import llm_call

class PositioningState(TypedDict):
    user_info: str
    profile: str
    competitors: str
    plan: str

#   节点  
def node_analyze(state: PositioningState) -> dict:
    prompt = f"""你是自媒体定位分析师。分析以下用户的画像：

【用户信息】
{state["user_info"]}

从4个维度分析：
1. 专业优势（核心竞争力）
2. 内容风格匹配（性格适合什么风格）
3. 3个推荐赛道（按优先级，说明理由）
4. 差异化切入点（与同类账号的区别）
"""
    return {"profile": llm_call(prompt, temperature=0.7)}

def node_competitors(state: PositioningState) -> dict:
    prompt = f"""基于以下用户画像，推荐5个对标账号：

{state["profile"]}

要求：
- 2个头部大号（天花板参考），3个腰部账号（可追赶目标）
- 每个账号说明：账号名、粉丝量级、内容特色、可借鉴点
"""
    return {"competitors": llm_call(prompt, temperature=0.7)}

def node_plan(state: PositioningState) -> dict:
    prompt = f"""基于以下信息，生成完整的账号定位策划方案：

【用户画像】
{state["profile"]}

【对标账号】
{state["competitors"]}

输出结构化方案：
1. **一句话定位**（你是谁，提供什么价值）
2. **目标受众**（3类核心粉丝画像）
3. **内容版图**（5个内容系列，每个说明主题+形式+频率）
4. **人设打造**（语言风格、视觉风格、记忆点）
5. **变现路径**（3条商业模式，从近到远排列）
6. **30天起号计划**（每周运营重点和内容数量）
"""
    return {"plan": llm_call(prompt, temperature=0.7)}

#   构建图  
builder = StateGraph(PositioningState)
builder.add_node("analyze", node_analyze)
builder.add_node("competitors", node_competitors)
builder.add_node("plan", node_plan)

builder.add_edge(START, "analyze")
builder.add_edge("analyze", "competitors")
builder.add_edge("competitors", "plan")
builder.add_edge("plan", END)

positioning_graph = builder.compile()

#   对外接口  
def run_positioning(user_info: str) -> dict:
    """运行定位工作流并保存结果"""
    result = positioning_graph.invoke({"user_info": user_info})

    # 保存到数据库
    try:
        with get_session() as session:
            record = PositioningRecord(
                user_name=user_info[:50],
                user_input=user_info,
                profile_analysis=result.get("profile", ""),
                positioning_plan=result.get("plan", ""),
            )
            session.add(record)
            session.commit()
    except Exception as e:
        print(f"[DB] 保存失败: {e}")

    return result

"""show_positioning页面"""
import streamlit as st

def _add_history(action: str, summary: str):
    """记录操作历史"""
    from datetime import datetime
    import streamlit as st
    st.session_state.history.append({
        "time": datetime.now().strftime("%H:%M"),
        "action": action,
        "summary": summary[:200],
    })

def show_positioning():
    """账号定位"""
    st.title("🎯 账号定位智能体")
    st.markdown("输入你的背景信息，AI帮你制定专属的账号定位策划方案")

    col1, col2 = st.columns(2)
    with col1:
        job = st.text_input("你的职业", placeholder="程序员 / 设计师 / 运营 / 学生...")
        skills = st.text_area("核心技能", placeholder="Python开发、视频剪辑、公开演讲...",
                               height=100)
    with col2:
        interests = st.text_area("兴趣领域", placeholder="科技测评、美食探店、知识分享...",
                                  height=100)
        platform = st.selectbox("目标平台", ["抖音", "小红书", "B站", "视频号", "全平台"])

    if st.button("🚀 生成定位方案", type="primary", use_container_width=True):
        if not job and not skills:
            st.warning("请至少填写职业或技能再生成")
            return

        user_input = f"职业：{job}\n技能：{skills}\n兴趣：{interests}\n目标平台：{platform}"

        with st.spinner("AI 正在分析你的画像..."):
            from workflows.positioning import run_positioning
            result = run_positioning(user_input)

        # 存入 session_state 防止下载按钮触发 rerun 后丢失
        st.session_state["pos_result"] = result
        st.session_state["pos_input"] = user_input

    #   从 session_state 恢复结果  
    result = st.session_state.get("pos_result")
    if result:
        st.success("✅ 方案生成完成！")

        tab1, tab2, tab3 = st.tabs(["📊 画像分析", "🔍 对标账号", "📝 完整方案"])
        with tab1:
            st.markdown(result.get("profile", "分析中..."))
        with tab2:
            st.markdown(result.get("competitors", "搜索中..."))
        with tab3:
            plan = result.get("plan", "")
            st.markdown(plan)
            st.download_button("📥 下载方案", plan,
                               file_name="账号定位方案.md",
                               mime="text/markdown")

        _add_history("账号定位", st.session_state.get("pos_input", ""))

> **关键设计**：结果存入 `session_state`，下载按钮触发 Streamlit rerun 时从缓存恢复，页面不会变空白。

#  平台 ID 映射 — 来自 TrendRadar config.yaml

# 默认 API（可自部署 newsnow）
DEFAULT_API_URL = "https://newsnow.busiyi.world/api/s"

# 默认请求头
DEFAULT_HEADERS = {
    "User-Agent": (
        "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
        "AppleWebKit/537.36 (KHTML, like Gecko) "
        "Chrome/91.0.4472.124 Safari/537.36"
    ),
    "Accept": "application/json, text/plain, */*",
    "Accept-Language": "zh-CN,zh;q=0.9,en;q=0.8",
}

class TrendRadarClient:
    """
    热点抓取客户端 — 基于 TrendRadar 的 DataFetcher 架构。

    使用 NewsNow 聚合 API 获取多平台热点榜单，
    支持自定义 API 地址（可自部署 newsnow）。

    使用方式:
        client = TrendRadarClient()
        topics = client.fetch_hot_topics(["douyin", "weibo", "zhihu"])
        # topics: [{"title": "...", "url": "...", "platform": "抖音", "rank": 1}, ...]
    """

    def __init__(self, api_url: str = None, proxy_url: str = None):
        """
        Args:
            api_url: NewsNow API 地址，默认使用公共 API
            proxy_url: 代理地址（可选）
        """
        self.api_url = api_url or DEFAULT_API_URL
        self.proxy_url = proxy_url

    #  单平台抓取
    def fetch_platform(
        self,
        platform_id: str,
        max_retries: int = 2,
    ) -> List[Dict]:
        """
        抓取单个平台的热点数据。

        Args:
            platform_id: 平台 ID（如 "douyin", "weibo"）
            max_retries: 最大重试次数

        Returns:
            热点列表 [{"title": str, "url": str, "rank": int}, ...]
        """
        url = f"{self.api_url}?id={platform_id}&latest"
        proxies = {"http": self.proxy_url, "https": self.proxy_url} if self.proxy_url else None
        platform_name = PLATFORM_IDS.get(platform_id, platform_id)

        for attempt in range(max_retries + 1):
            try:
                resp = requests.get(
                    url, proxies=proxies, headers=DEFAULT_HEADERS, timeout=10
                )
                resp.raise_for_status()
                data = resp.json()

                status = data.get("status", "")
                if status not in ("success", "cache"):
                    raise ValueError(f"API 状态异常: {status}")

                items = data.get("items", [])
                topics = []
                for i, item in enumerate(items, 1):
                    title = item.get("title")
                    if not title or not str(title).strip():
                        continue
                    topics.append({
                        "title": str(title).strip(),
                        "url": item.get("url", ""),
                        "mobile_url": item.get("mobileUrl", ""),
                        "platform": platform_name,
                        "platform_id": platform_id,
                        "rank": i,
                        "heat": self._estimate_heat(i, len(items)),
                    })

                status_info = "最新" if status == "success" else "缓存"
                print(f"[TrendRadar] {platform_name} 获取成功（{status_info}，{len(topics)} 条）")
                return topics

            except Exception as e:
                if attempt < max_retries:
                    wait = random.uniform(2, 5)
                    print(f"[TrendRadar] {platform_name} 失败，{wait:.1f}s 后重试: {e}")
                    time.sleep(wait)
                else:
                    print(f"[TrendRadar] {platform_name} 最终失败: {e}")
                    return []

        return []

    #  多平台批量抓取
    def fetch_hot_topics(
        self,
        platform_ids: List[str] = None,
        request_interval: float = 0.3,
    ) -> Dict[str, List[Dict]]:
        """
        批量抓取多个平台的热点。

        Args:
            platform_ids: 平台 ID 列表，默认抓取主流 6 个平台
            request_interval: 请求间隔（秒）

        Returns:
            {平台ID: [热点列表], ...}
        """
        if platform_ids is None:
            platform_ids = ["douyin", "weibo", "zhihu",
                           "bilibili-hot-search", "toutiao", "baidu"]

        results = {}
        for i, pid in enumerate(platform_ids):
            topics = self.fetch_platform(pid)
            if topics:
                results[pid] = topics

            # 请求间隔（最后一个不需要等待）
            if i < len(platform_ids) - 1:
                jitter = request_interval + random.uniform(-0.1, 0.2)
                time.sleep(max(0.1, jitter))

        total = sum(len(v) for v in results.values())
        print(f"[TrendRadar] 批量抓取完成: {len(results)}/{len(platform_ids)} 平台，共 {total} 条")
        return results

    #  适配 self_media_agent 工作流格式
    def fetch_for_workflow(
        self,
        platform: str = "全部",
        platform_ids: List[str] = None,
    ) -> List[Dict]:
        """
        抓取热点并转换为工作流所需格式。

        Args:
            platform: 中文平台名筛选（"抖音"/"小红书"/"全部"）
            platform_ids: 直接指定平台 ID 列表

        Returns:
            [{"source": "抖音", "title": "...", "heat": 980000, "url": "..."}, ...]
        """
        # 平台名 → ID 映射
        name_to_ids = {
            "抖音": ["douyin"],
            "小红书": ["xiaohongshu"],
            "微博": ["weibo"],
            "知乎": ["zhihu"],
            "B站": ["bilibili-hot-search"],
            "今日头条": ["toutiao"],
            "百度": ["baidu"],
            "全部": ["douyin", "weibo", "zhihu", "bilibili-hot-search",
                    "toutiao", "baidu", "xiaohongshu"],
        }

        if platform_ids is None:
            platform_ids = name_to_ids.get(platform, ["douyin", "weibo", "zhihu"])

        all_topics = []
        for pid in platform_ids:
            topics = self.fetch_platform(pid)
            for t in topics:
                all_topics.append({
                    "source": t["platform"],
                    "title": t["title"],
                    "heat": t["heat"],
                    "url": t["url"],
                    "platform_id": t["platform_id"],
                    "rank": t["rank"],
                })

        # 按热度降序排列
        all_topics.sort(key=lambda x: x["heat"], reverse=True)
        return all_topics

    #  辅助方法
    @staticmethod
    def _estimate_heat(rank: int, total: int) -> int:
        """根据排名估算热度值（排名越前热度越高）"""
        base = 1000000
        ratio = 1.0 - (rank - 1) / max(total, 1)
        return int(base * ratio)

    @staticmethod
    def get_available_platforms() -> Dict[str, str]:
        """获取所有可用平台"""

    小红书分享格式示例:
        "57 【标题】 emoji ID emoji https://www.xiaohongshu.com/..."

        提取到的第一个 HTTP(S) URL，未找到则返回原文本

    if text.startswith("http://") or text.startswith("https://"):
        return text

    # 匹配 HTTP(S) URL
    url_pattern = r'https?://[^\s一-鿿　-〿＀-￯]+'
    matches = re.findall(url_pattern, text)
    if matches:
        url = matches[0].rstrip(".,;:!?）)】』\"'")
        return url

    return text

async def _generate_tts_async(text: str, output_path: str) -> bool:
    """Edge TTS 异步生成（内部实现）"""
    import edge_tts

    # 截断过长文本

def generate_tts(text: str, output_path: str = None) -> str:
    """
    使用 Edge TTS 生成语音配音。

    Args:
        text: 要合成语音的文本

        音频文件路径，失败返回空字符串

def video_to_audio(
    input_video: str,
    output_audio: str = None,
    bitrate: str = "192k",
    fps: int = 44100,
) -> str:
    """
    从视频中提取音频（使用 FFmpeg）。

    Args:
        input_video: 输入视频文件路径
        output_audio: 输出音频路径（默认与视频同目录，.mp3 后缀）
        bitrate: 音频码率（默认 192k）
        fps: 采样率（默认 44100 Hz）

    Returns:

        if os.path.exists(output_audio) and os.path.getsize(output_audio) > 0:
            print(f"[video_to_audio] [OK] {output_audio}")

    except Exception as e:

"""FunASR WebSocket 服务器：接收音频字节流，返回转写文本（可选句级时间戳）。

def _split_sentences(text: str, timestamp: list) -> list:
    """按标点把带时间戳的转写结果聚合为句级 segments。

    词级 timestamp 与文本中的非标点字符一一对应（中文逐字、英文按 token）；
    遍历文本，标点依附当前句并触发切句，非标点字符消耗一个 timestamp。
    若消耗数量与 timestamp 长度不一致（英文子词等边界情况），
    降级为整段返回。

    Args:
        text: 带标点的转写文本。
        timestamp: 词级时间戳 [[start_ms, end_ms], ...]。

    Returns:
        [{"text": str, "start": int, "end": int}, ...]，毫秒单位。
    """
    sentences = []
    cur_chars: list[str] = []
    cur_start: int | None = None
    cur_end: int | None = None
    tok_idx = 0

    def flush() -> None:
        """结束当前句并追加到结果列表。"""
        nonlocal cur_chars, cur_start, cur_end
        seg_text = "".join(cur_chars).strip()
        if seg_text and cur_start is not None:
            sentences.append(
                {"text": seg_text, "start": int(cur_start), "end": int(cur_end)}
            )
        cur_chars, cur_start, cur_end = [], None, None

    for ch in text:
        if ch in _SENT_END_CHARS:
            cur_chars.append(ch)
            flush()
            continue
        if ch.isspace():
            continue
        if tok_idx >= len(timestamp):
            # 文本比时间戳长（异常），放弃对齐
            logger.warning("文本与 timestamp 数量不匹配，降级为整段返回")
            return [{"text": text.strip(), "start": timestamp[0][0],
                     "end": timestamp[-1][-1]}] if timestamp else []
        start, end = timestamp[tok_idx]
        tok_idx += 1
        if cur_start is None:
            cur_start = start
        cur_end = end
        cur_chars.append(ch)
    flush()

    if tok_idx != len(timestamp):
        logger.warning(
            "消耗 token %d 与 timestamp %d 不一致，降级为整段返回",
            tok_idx, len(timestamp),
        )
        if timestamp:
            return [{"text": text.strip(), "start": timestamp[0][0],
                     "end": timestamp[-1][-1]}]
        return []
    return sentences

def _normalize_sentences(item: dict) -> list:
    """将 FunASR 原生 sentence_info 转为对外句段格式。

    Args:
        item: FunASR 单条识别结果，包含 sentence_info 或 timestamp。

    Returns:
        [{"text": str, "start": int, "end": int}, ...]；原生句段不可用时
        回退到词级时间戳切分。
    """
    sentences = []
    for sentence in item.get("sentence_info") or []:
        text = str(sentence.get("text") or "").strip()
        start = sentence.get("start")
        end = sentence.get("end")
        if text and start is not None and end is not None and start < end:
            sentences.append({
                "text": text,
                "start": int(start),
                "end": int(end),
            })
    if sentences:
        return sentences
    return _split_sentences(
        str(item.get("text") or "").strip(), item.get("timestamp") or []
    )

async def handle_connection(ws) -> None:
    """处理一条 WebSocket 连接：收音频 -> 转写 -> 回文本（可选句级时间戳）。

    Args:
        ws: websockets 连接对象。
    """
    peer = ws.remote_address
    logger.info("客户端连接: %s", peer)
    audio = bytearray()
    wav_format = "wav"
    want_ts = False
    try:
        async for message in ws:
            if isinstance(message, bytes):
                audio.extend(message)
                continue
            data = json.loads(message)
            wav_format = data.get("wav_format", wav_format)
            if data.get("return_timestamps"):
                want_ts = True
            if data.get("is_speaking") is False:
                break  # 音频结束，开始转写
    except websockets.ConnectionClosed:
        logger.warning("客户端 %s 连接中断", peer)
        return

    logger.info("收到音频 %.1f KB（时间戳=%s），开始转写", len(audio) / 1024, want_ts)
    payload = await transcribe(bytes(audio), wav_format, want_ts)
    await ws.send(json.dumps(payload, ensure_ascii=False))
    logger.info(
        "转写完成（%d 字，%d 句），已返回客户端 %s",
        len(payload.get("text", "")), len(payload.get("sentences", [])), peer,
    )
    await ws.close()

async def transcribe(audio_bytes: bytes, wav_format: str,
                     want_ts: bool = False) -> dict:
    """调用 FunASR 模型转写音频字节流。

    Args:
        audio_bytes: 音频文件完整字节。
        wav_format: 音频格式后缀（wav/mp3 等）。
        want_ts: 是否返回句级时间戳（走 seaco-paraformer 通道）。

    Returns:
        {"mode": "offline", "text": str, "is_final": True,
         "sentences": [...]?}；失败时 text 为空字符串。
    """
    suffix = f".{wav_format or 'wav'}"
    payload = {"mode": "offline", "text": "", "is_final": True}
    tmp_path = None
    try:
        with tempfile.NamedTemporaryFile(suffix=suffix, delete=False) as tmp:
            tmp.write(audio_bytes)
            tmp_path = tmp.name

        if want_ts:
            m = _get_ts_model()
            result = await asyncio.to_thread(
                m.generate,
                input=tmp_path,
                batch_size_s=300,
                sentence_timestamp=True,
            )
            if isinstance(result, list) and result:
                item = result[0]
                payload["text"] = str(item.get("text") or "").strip()
                sentences = _normalize_sentences(item)
                if sentences:
                    payload["sentences"] = sentences
            return payload

        result = await asyncio.to_thread(model.generate, input=tmp_path)
        if isinstance(result, list) and result:
            payload["text"] = (result[0].get("text") or "").strip()
    except Exception as exc:  # noqa: BLE001 - 单请求失败不影响服务
        logger.exception("转写失败: %s", exc)
    finally:
        if tmp_path:
            Path(tmp_path).unlink(missing_ok=True)
    return payload

async def main(port: int) -> None:
    """启动 WebSocket 服务。

    Args:
        port: 监听端口。
    """
    load_model()
    async with websockets.serve(handle_connection, "0.0.0.0", port, max_size=None):
        logger.info("FunASR WebSocket 服务已启动: 0.0.0.0:%d", port)
        await asyncio.Future()  # 永久运行

if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--port", type=int, default=6006)
    args = parser.parse_args()
    asyncio.run(main(args.port))

# 文件: workflows/replicate.py
"""内容复刻工作流 — 提取 → 分析 → 仿写 → 标题"""
from typing import TypedDict
from langgraph.graph import StateGraph, START, END
from workflows import llm_call
from tools.media_tools import download_video, extract_audio_text

class ReplicateState(TypedDict):
    source_url: str
    original_text: str
    viral_analysis: str
    rewritten: str
    titles: str

#   节点  
def node_extract(state: ReplicateState) -> dict:
    """提取视频文案"""
    url = state["source_url"]
    path = download_video(url)
    text = extract_audio_text(path)
    return {"original_text": text}

def node_analyze(state: ReplicateState) -> dict:
    """分析爆款要素"""
    prompt = f"""拆解以下爆款文案的结构：

{state["original_text"]}

从以下维度分析：
1. **钩子类型**（前3句用了什么技巧：悬念/反差/痛点/利益）
2. **结构模板**（可复用的文案框架）
3. **情绪节奏**（每段的情绪高低点）
4. **金句亮点**（最容易引发共鸣的句子）
5. **互动引导**（点赞/评论/关注的引导方式）
"""
    return {"viral_analysis": llm_call(prompt, temperature=0.6)}

def node_rewrite(state: ReplicateState) -> dict:
    """仿写新文案"""
    prompt = f"""根据以下爆款分析，仿写一篇新文案：

【爆款要素分析】
{state["viral_analysis"]}

【参考原文】
{state["original_text"]}

要求：
- 保持相同结构模板和节奏
- 换成新主题（同类不同题）
- 保留钩子、转折、金句、互动引导
- 字数控制在原文的80%-120%
"""
    return {"rewritten": llm_call(prompt, temperature=0.8)}

def node_titles(state: ReplicateState) -> dict:
    """生成标题"""
    prompt = f"""为以下文案生成5个爆款标题：

{state["rewritten"]}

5种类型各一个：
1. 数字型（"3个方法..."）
2. 疑问型（"为什么..."）
3. 痛点型（"别再..."）
4. 悬念型（"原来..."）
5. 命令型（"一定要..."）
每个标题不超过25字。
"""
    return {"titles": llm_call(prompt, temperature=0.8)}

#   构建图  
builder = StateGraph(ReplicateState)
builder.add_node("extract", node_extract)
builder.add_node("analyze", node_analyze)
builder.add_node("rewrite", node_rewrite)
builder.add_node("titles", node_titles)

builder.add_edge(START, "extract")
builder.add_edge("extract", "analyze")
builder.add_edge("analyze", "rewrite")
builder.add_edge("rewrite", "titles")
builder.add_edge("titles", END)

replicate_graph = builder.compile()

def run_replicate(url: str) -> dict:
    """运行内容复刻工作流"""
    return replicate_graph.invoke({"source_url": url})

# 文件: tools/heygem_client.py（关键接口）
"""
Duix Avatar (HeyGem) 数字人 API 客户端
实现模特视频声音提取与克隆、克隆声音 TTS、静音视频生成、Docker 路径转换，以及数字人任务的提交和进度查询

API 端点:
  8383: POST /easy/submit  — 提交视频生成任务
  8383: GET  /easy/query   — 查询任务进度
  18180: POST /v1/preprocess_and_tran — 声音克隆预处理
  18180: POST /v1/invoke              — 克隆声音 TTS 合成

Docker 部署:
  docker pull guiji2025/heygem.ai
  # 三个容器: gen-video(8383) / tts(18180) / asr(10095)

配置在 config.py → .env:
  HEYGEM_HOST=http://localhost:8383
  HEYGEM_TTS_HOST=http://localhost:18180
  HEYGEM_DATA_DIR=c:/duix_avatar_data/face2face
"""
import os
import time
import uuid
import shutil
import subprocess
import httpx
from config import settings

# 从统一配置读取（全部来自 .env / config.py，不再用 os.getenv ——
# 因为 os.getenv 读不到 .env 文件，.env 只被 pydantic-settings 加载进 Settings）
HEYGEM_HOST = settings.heygem_host
HEYGEM_TTS_HOST = settings.heygem_tts_host
HEYGEM_DATA_DIR = settings.heygem_data_dir
# TTS 服务有独立的挂载目录（与 gen-video 不同！）
HEYGEM_TTS_DATA_DIR = settings.get_heygem_tts_data_dir()

#  声音克隆 + 克隆声音 TTS
def heygem_voice_clone(source_video: str, lang: str = "zh") -> dict:
    """
    从模特视频中提取声音并克隆。

    流程：
    1. FFmpeg 提取视频音轨 → wav
    2. 复制到 TTS 服务数据目录
    3. 调用 /v1/preprocess_and_tran 进行声音预处理

    Args:
        source_video: 模特视频路径
        lang: 语言（zh/en/ja/ko）

    Returns:
        {"success": bool, "asr_format_audio_url": str,
         "reference_audio_text": str, "message": str}
    """
    if not os.path.exists(source_video):
        return {"success": False, "message": "视频文件不存在"}

    # 1. 提取音频为 wav → 必须放到 TTS 服务的挂载目录！
    audio_name = f"voice_{uuid.uuid4().hex[:8]}.wav"
    os.makedirs(HEYGEM_TTS_DATA_DIR, exist_ok=True)
    audio_path = os.path.join(HEYGEM_TTS_DATA_DIR, audio_name)

    try:
        subprocess.run([
            "ffmpeg", "-y", "-i", source_video,
            "-vn", "-acodec", "pcm_s16le",
            "-ar", "16000", "-ac", "1",
            audio_path,
        ], check=True, capture_output=True, timeout=30)

        if not os.path.exists(audio_path) or os.path.getsize(audio_path) < 1024:
            return {"success": False, "message": "音频提取失败或文件过小"}
    except Exception as e:
        return {"success": False, "message": f"音频提取异常: {e}"}

    # 2. TTS API 需要相对路径（相对于 /code/data/ 即 voice/data 目录）
    container_audio = audio_name

    # 3. 调用声音预处理 API
    try:
        print(f"[HeyGem-Voice] 声音克隆预处理: {container_audio}")
        resp = httpx.post(
            f"{HEYGEM_TTS_HOST}/v1/preprocess_and_tran",
            json={
                "format": "wav",
                "reference_audio": container_audio,
                "lang": lang,
            },
            timeout=60,
        )
        data = resp.json()
        # 处理可能返回 list（验证错误）的情况
        if isinstance(data, list):
            error_msg = data[0].get("msg", str(data)) if data else "未知错误"
            return {"success": False, "message": f"API 返回错误: {error_msg}"}

        asr_url = data.get("asr_format_audio_url", "")
        ref_text = data.get("reference_audio_text", "")

        if asr_url and ref_text:
            print(f"[HeyGem-Voice] 声音克隆成功: text={ref_text[:50]}...")
            return {
                "success": True,
                "asr_format_audio_url": asr_url,
                "reference_audio_text": ref_text,
                "message": "声音克隆成功",
            }

        # 常见错误处理
        msg = data.get("msg", "")
        if "asr failed" in str(msg).lower():
            return {"success": False,
                    "message": "ASR 识别失败。请确保模特视频中有人声且语音清晰、无背景音乐。"}
        if "file not exists" in str(msg).lower():
            return {"success": False,
                    "message": f"音频文件未找到。请检查 TTS 数据目录: {HEYGEM_TTS_DATA_DIR}"}
        return {"success": False, "message": f"预处理失败: {msg}"}

    except httpx.ConnectError:
        return {"success": False, "message": f"TTS 服务 ({HEYGEM_TTS_HOST}) 不可达"}
    except Exception as e:
        return {"success": False, "message": f"声音克隆异常: {e}"}

def heygem_tts_with_cloned_voice(
    text: str,
    asr_format_audio_url: str,
    reference_audio_text: str,
    output_path: str = None,
) -> str:
    """
    使用克隆的声音合成 TTS 音频。

    Args:
        text: 要合成语音的文本
        asr_format_audio_url: 声音克隆返回的预处理音频路径
        reference_audio_text: 声音克隆返回的参考文本
        output_path: 输出音频路径

    Returns:
        wav 音频文件路径，失败返回空字符串
    """
    if output_path is None:
        # TTS 输出放到 gen-video 的共享目录，方便后续视频合成
        output_path = os.path.join(HEYGEM_DATA_DIR,
                                   f"tts_cloned_{uuid.uuid4().hex[:8]}.wav")

    try:
        speaker_id = str(uuid.uuid4())
        print(f"[HeyGem-TTS] 合成: {text[:50]}...")

        resp = httpx.post(
            f"{HEYGEM_TTS_HOST}/v1/invoke",
            json={
                "speaker": speaker_id,
                "text": text,
                "format": "wav",
                "topP": 0.7,
                "max_new_tokens": 1024,
                "chunk_length": 100,
                "repetition_penalty": 1.2,
                "temperature": 0.7,
                "need_asr": False,
                "streaming": False,
                "is_fixed_seed": 0,
                "is_norm": 0,
                "reference_audio": asr_format_audio_url,
                "reference_text": reference_audio_text,
            },
            timeout=120,
        )

        if resp.status_code == 200:
            # 响应是音频二进制数据
            content_type = resp.headers.get("content-type", "")
            if "audio" in content_type or "octet-stream" in content_type:
                os.makedirs(os.path.dirname(output_path), exist_ok=True)

def _to_container_path(host_path: str) -> str:
    """
    将宿主机路径转换为容器内路径。

    容器挂载: /code/data → 宿主机 HEYGEM_DATA_DIR
    如果文件不在挂载目录下，复制过去。
    """
    # 如果已经是容器路径格式，直接返回
    if host_path.startswith("/code/"):
        return host_path

    abs_path = os.path.abspath(host_path)
    data_dir = os.path.abspath(HEYGEM_DATA_DIR)

    # 在挂载目录下 → 转换为容器路径
    if abs_path.startswith(data_dir):
        rel = os.path.relpath(abs_path, data_dir)
        return f"/code/data/{rel.replace(os.sep, '/')}"

    # 不在挂载目录 → 复制到挂载目录
    os.makedirs(data_dir, exist_ok=True)
    basename = os.path.basename(abs_path)
    dst = os.path.join(data_dir, basename)
    if abs_path != dst:
        import shutil
        shutil.copy2(abs_path, dst)
        print(f"[HeyGem] 文件已复制到共享目录: {dst}")
    return f"/code/data/{basename}"

# 允许的媒体格式
VIDEO_EXTS = {".mp4", ".mov", ".avi", ".webm"}
IMAGE_EXTS = {".png", ".jpg", ".jpeg", ".gif", ".bmp", ".webp"}

def _list_avatars() -> list[dict]:
    """扫描 HeyGem 数据目录，列出所有数字人模特文件

    Returns:
        [{"path": str, "name": str, "type": "video"|"image", "size_mb": float}, ...]
    """
    avatars = []
    if not os.path.isdir(HEYGEM_DATA_DIR):
        return avatars

    seen = set()
    for f in sorted(os.listdir(HEYGEM_DATA_DIR), key=lambda x: os.path.getmtime(
            os.path.join(HEYGEM_DATA_DIR, x)), reverse=True):
        full = os.path.join(HEYGEM_DATA_DIR, f)
        if not os.path.isfile(full):
            continue
        name, ext = os.path.splitext(f)
        ext = ext.lower()

        # 跳过非媒体文件
        if ext not in VIDEO_EXTS and ext not in IMAGE_EXTS:
            continue
        # 跳过 muted 版本（静音视频是衍生物）
        if name.endswith("_muted"):
            continue
        # 跳过临时/中间文件
        if any(p in full.lower() for p in ["/temp/", "/log/", "/result/",
                                             "voice_", "voiceover_", "voice_cloned",
                                             "test_", "tts_", "long_tts"]):
            continue

        base = name  # 去掉 _muted 后的名字，用于去重
        if base in seen:
            continue
        seen.add(base)

        size_mb = os.path.getsize(full) / (1024 * 1024)
        avatars.append({
            "path": full,
            "name": f,
            "type": "video" if ext in VIDEO_EXTS else "image",
            "size_mb": size_mb,
        })

    return avatars

def normalize_path(p: str) -> str:
    """修复 Git Bash / Windows 路径混用导致的各种畸形路径。

    常见畸形及修复：
      C:\\c\\Users\\...  →  C:\\Users\\...
      C:/c/Users/...    →  C:\\Users\\...
      /c/Users/...      →  C:\\Users\\...
      /d/project/...    →  D:\\project\\...
      C:\\d\\Data\\...    →  D:\\Data\\... (当 C: 后面的盘符字母与 C: 不同时)
    """
    if not p or not isinstance(p, str):
        return p

    original = p

    # 1) 修复 Windows 盘符 + Git Bash 盘符前缀叠加：
    #    C:\\c\\Users\\... → C:\\Users\\...（盘符字母相同）
    #    C:\\d\\Data\\...  → D:\\Data\\...（盘符字母不同，以第二个为准）
    #    匹配: <盘符1>:\ 或 :/  然后 <盘符2>\ 或 /
    m = re.match(r'^([A-Za-z]):[\\/]([a-z])([\\/].+)$', p)
    if m:
        drive1 = m.group(1).lower()
        drive2 = m.group(2).lower()
        rest = m.group(3)
        if drive1 == drive2:
            # 同一个盘符重复：C:\c\Users → C:\Users
            p = f'{m.group(1)}:{rest}'
        else:
            # 不同盘符：可能是 LLM 把目标盘符误拼成 "当前盘符:\目标盘符\..."
            p = f'{m.group(2).upper()}:{rest}'

    # 2) 修复纯 Git Bash 路径：/c/Users/... → C:\\Users\\...
    if sys.platform == 'win32':
        m2 = re.match(r'^/([a-zA-Z])/(.+)$', p)
        if m2:
            drive = m2.group(1).upper()
            rest = m2.group(2)
            p = f'{drive}:\\{rest}'

    # 3) 统一斜杠方向（Windows）
    if sys.platform == 'win32' and '\\' not in p and '/' in p:
        p = p.replace('/', '\\')
        # 修复 /videos、/videos/output.mp4 → \videos\... 被 Windows 误判为 C:\videos\...
        # 原始输入以 / 开头但不是 /c/ 格式（Git Bash 路径）→ 去掉前导反斜杠，恢复为相对路径
        if original.startswith('/') and not re.match(r'^/[a-zA-Z]/', original):
            p = p.lstrip('\\')

    # 4) 最终规范化
    try:
        p = os.path.normpath(p)
    except Exception:
        pass

    if p != original:
        print(f"[PathFix] 路径已修复: {original!r} → {p!r}")
    return p
